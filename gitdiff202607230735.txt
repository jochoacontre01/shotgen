+def _preload_nvhpc_libs():
+    """
+    Pre-load NVHPC OpenACC runtime libraries with RTLD_GLOBAL so OpenACC symbols
+    are claimed in global scope before third-party modules (like PyLops/PyTorch)
+    pull GNU libgomp.so.1 into process memory.
+    """
+    nvc_bin = shutil.which("nvc++") or shutil.which("nvc")
+    if not nvc_bin:
+        return
+    nv_libdir = os.path.abspath(os.path.join(os.path.dirname(nvc_bin), "..", "lib"))
+    if os.path.isdir(nv_libdir):
+        for libname in ["libnvc.so", "libnvcpumath.so", "libnvomp.so", "libacchost.so"]:
+            libpath = os.path.join(nv_libdir, libname)
+            if os.path.isfile(libpath):
+                try:
+                    ctypes.CDLL(libpath, mode=ctypes.RTLD_GLOBAL)
+                except Exception:
+                    pass
